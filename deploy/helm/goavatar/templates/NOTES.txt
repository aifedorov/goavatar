Install or upgrade:
  helm upgrade --install goavatar {{ .Chart.Name }} -n {{ include "goavatar.namespace" . }} --create-namespace -f values-local.yaml

Check workloads:
  kubectl get pods -n {{ include "goavatar.namespace" . }}
  kubectl get jobs -n {{ include "goavatar.namespace" . }}

Port-forward app:
  kubectl port-forward -n {{ include "goavatar.namespace" . }} svc/goavatar-server 8080:80

Health:
  curl http://localhost:8080/health

Metrics:
  curl http://localhost:8080/metrics
